0241:                 HideHint();

0463:     private Transform FindEmptySpotUnderPointer(PointerEventData eventData)
0465:         if (EventSystem.current == null || gm == null || gm.EmptySpot == null) return null;
0466: 
0467:         var results = new List<RaycastResult>();
0468:         EventSystem.current.RaycastAll(eventData, results);
0469:         foreach (var res in results)
0471:             var t = res.gameObject.transform;
0472:             while (t != null)
0474:                 if (t.gameObject.name == gm.EmptySpot.name)
0475:                     return t;
